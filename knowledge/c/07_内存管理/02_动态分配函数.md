# 动态分配函数

## malloc — 分配未初始化的内存

### 基本语法

```c
void *malloc(size_t size);
```

### 功能

在堆上分配指定字节数的内存，返回指向分配内存的指针。分配的内存**未初始化**，内容是随机的。

### 示例

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 10;

    // 分配 n 个 int 的内存
    int *arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        printf("内存分配失败\n");
        return 1;
    }

    // 使用内存
    for (int i = 0; i < n; i++) {
        arr[i] = i * 10;
    }

    // 打印
    for (int i = 0; i < n; i++) {
        printf("arr[%d] = %d\n", i, arr[i]);
    }

    // 释放内存
    free(arr);
    arr = NULL;

    return 0;
}
```

### 注意事项

1. **返回值类型是 `void*`**，需要强制转换
2. **分配失败返回 NULL**
3. **分配的内存未初始化**
4. **必须手动释放**

## calloc — 分配并初始化为零的内存

### 基本语法

```c
void *calloc(size_t num, size_t size);
```

### 功能

在堆上分配 `num` 个大小为 `size` 的内存块，返回指向分配内存的指针。分配的内存**自动初始化为 0**。

### 示例

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 10;

    // 分配 n 个 int 的内存，并初始化为 0
    int *arr = (int *)calloc(n, sizeof(int));
    if (arr == NULL) {
        printf("内存分配失败\n");
        return 1;
    }

    // 所有元素都是 0
    for (int i = 0; i < n; i++) {
        printf("arr[%d] = %d\n", i, arr[i]);  // 输出 0
    }

    free(arr);
    arr = NULL;

    return 0;
}
```

### malloc vs calloc

| 特性 | malloc | calloc |
|------|--------|--------|
| 参数 | 一个参数（总字节数） | 两个参数（元素个数和元素大小） |
| 初始化 | 不初始化（垃圾值） | 初始化为 0 |
| 返回值 | void* | void* |
| 用途 | 需要手动初始化时 | 需要零初始化时 |

### malloc/calloc/realloc/free 对比

| 函数 | 功能 | 失败时返回 | 使用注意 |
|------|------|-----------|----------|
| `malloc(size)` | 分配 size 字节，不初始化 | NULL | 返回 `void*`，需强转后使用 |
| `calloc(n, size)` | 分配 n×size 字节，并清零 | NULL | 大块分配时可能比 malloc+memset 更快 |
| `realloc(p, size)` | 调整已分配内存的大小 | NULL（原内存仍有效） | 不要用 `p = realloc(p,...)` 直接覆盖 |
| `free(p)` | 释放 p 指向的内存 | 无返回值 | 只能释放堆内存，释放后置 NULL |

## realloc — 重新分配内存

### 基本语法

```c
void *realloc(void *ptr, size_t new_size);
```

### 功能

调整已分配内存块的大小。如果原内存块后面有足够的空间，则直接扩展；否则分配新的内存块，复制数据，释放旧内存块。

### 示例

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    // 初始分配 5 个 int
    int *arr = (int *)malloc(5 * sizeof(int));
    if (arr == NULL) {
        printf("内存分配失败\n");
        return 1;
    }

    for (int i = 0; i < 5; i++) {
        arr[i] = i * 10;
    }

    // 重新分配为 10 个 int
    int *new_arr = (int *)realloc(arr, 10 * sizeof(int));
    if (new_arr == NULL) {
        printf("重新分配失败\n");
        free(arr);
        return 1;
    }
    arr = new_arr;

    // 新元素未初始化
    for (int i = 5; i < 10; i++) {
        arr[i] = i * 10;
    }

    // 打印
    for (int i = 0; i < 10; i++) {
        printf("arr[%d] = %d\n", i, arr[i]);
    }

    free(arr);
    arr = NULL;

    return 0;
}
```

### realloc 的注意事项

1. **ptr 为 NULL 时，等价于 malloc**
   ```c
   int *p = realloc(NULL, 100);  // 等价于 malloc(100)
   ```

2. **new_size 为 0 时，释放内存并返回 NULL**
   ```c
   free(arr);
   arr = realloc(arr, 0);  // 释放内存，arr = NULL
   ```

3. **realloc 可能移动内存地址**
   ```c
   int *arr = malloc(5 * sizeof(int));
   int *new_arr = realloc(arr, 100 * sizeof(int));
   // 如果 new_arr != arr，原 arr 指向的内存已释放
   // 应该使用 new_arr，而不是 arr
   ```

4. **realloc 失败时，原内存块仍然有效**
   ```c
   int *arr = malloc(5 * sizeof(int));
   int *new_arr = realloc(arr, 1000000000);
   if (new_arr == NULL) {
       // arr 仍然有效，可以继续使用
       // 但应该释放 arr 并处理错误
       free(arr);
       return 1;
   }
   arr = new_arr;
   ```

5. **realloc 使原指针失效**
   ```c
   // 错误写法：直接覆盖原指针
   int *arr = malloc(10 * sizeof(int));
   arr = realloc(arr, 100 * sizeof(int));
   // 如果 realloc 失败返回 NULL，arr 就被赋成 NULL，
   // 原内存既无法访问也无法释放 → 内存泄漏

   // 正确写法：用临时指针接收结果
   int *arr = malloc(10 * sizeof(int));
   int *tmp = realloc(arr, 100 * sizeof(int));
   if (tmp == NULL) {
       free(arr);   // 原内存仍有效，先释放
       return 1;
   }
   arr = tmp;       // 成功后原 arr 已失效，改用 tmp
   ```

> **注意**：`realloc` 成功后，**原来的指针值就失效了**（可能被移动到了新地址），此后只能使用返回的新指针。

## free — 释放内存

### 基本语法

```c
void free(void *ptr);
```

### 功能

释放之前通过 malloc、calloc 或 realloc 分配的内存。

### 示例

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int *arr = (int *)malloc(10 * sizeof(int));
    if (arr != NULL) {
        // 使用内存...
        free(arr);
        arr = NULL;  // 释放后将指针设为 NULL
    }

    return 0;
}
```

### free 的注意事项

1. **只能释放堆上分配的内存**
   ```c
   int arr[10];
   // free(arr);  // 错误！arr 是栈上分配的
   ```

2. **不能重复释放**
   ```c
   int *arr = (int *)malloc(10 * sizeof(int));
   free(arr);
   // free(arr);  // 错误！重复释放
   ```

3. **释放后应将指针设为 NULL**
   ```c
   free(arr);
   arr = NULL;  // 避免野指针
   ```

4. **释放 NULL 指针是安全的**
   ```c
   free(NULL);  // 安全，什么也不做
   ```

## 动态内存分配完整示例

### 动态数组

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    int size;
    int capacity;
} DynamicArray;

DynamicArray *create_array(int initial_capacity) {
    DynamicArray *da = (DynamicArray *)malloc(sizeof(DynamicArray));
    if (da == NULL) {
        return NULL;
    }

    da->data = (int *)malloc(initial_capacity * sizeof(int));
    if (da->data == NULL) {
        free(da);
        return NULL;
    }

    da->size = 0;
    da->capacity = initial_capacity;
    return da;
}

void push_back(DynamicArray *da, int value) {
    if (da->size >= da->capacity) {
        int new_capacity = da->capacity * 2;
        int *new_data = (int *)realloc(da->data, new_capacity * sizeof(int));
        if (new_data == NULL) {
            return;  // 分配失败
        }
        da->data = new_data;
        da->capacity = new_capacity;
    }
    da->data[da->size++] = value;
}

void destroy_array(DynamicArray *da) {
    if (da != NULL) {
        free(da->data);
        free(da);
    }
}

int main(void) {
    DynamicArray *arr = create_array(4);
    if (arr == NULL) {
        return 1;
    }

    for (int i = 0; i < 10; i++) {
        push_back(arr, i * 10);
    }

    for (int i = 0; i < arr->size; i++) {
        printf("arr[%d] = %d\n", i, arr->data[i]);
    }

    destroy_array(arr);
    return 0;
}
```

### 动态字符串

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *read_line(void) {
    int capacity = 64;
    int length = 0;
    char *buffer = (char *)malloc(capacity);
    if (buffer == NULL) {
        return NULL;
    }

    int ch;
    while ((ch = getchar()) != '\n' && ch != EOF) {
        if (length + 1 >= capacity) {
            capacity *= 2;
            char *new_buffer = (char *)realloc(buffer, capacity);
            if (new_buffer == NULL) {
                free(buffer);
                return NULL;
            }
            buffer = new_buffer;
        }
        buffer[length++] = (char)ch;
    }
    buffer[length] = '\0';

    return buffer;
}

int main(void) {
    printf("请输入一行文本：");
    char *line = read_line();
    if (line != NULL) {
        printf("你输入的是：%s\n", line);
        free(line);
    }

    return 0;
}
```

### 结构体指针动态分配

动态分配结构体时，用 `malloc` 按结构体大小分配，返回的 `void*` 强转为结构体指针后，用 `->` 访问成员：

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char name[20];
    int score;
} Student;

int main(void) {
    // 动态分配一个 Student 结构体
    Student *stu = (Student *)malloc(sizeof(Student));
    if (stu == NULL) {
        printf("内存分配失败\n");
        return 1;
    }

    strcpy(stu->name, "Tom");
    stu->score = 95;

    printf("姓名：%s，成绩：%d\n", stu->name, stu->score);

    free(stu);       // 释放结构体
    stu = NULL;      // 置 NULL 防野指针
    return 0;
}
```

## 常见内存错误

### 1. 内存泄漏

```c
void leak_example(void) {
    int *arr = (int *)malloc(100 * sizeof(int));
    // 忘记 free(arr)
    return;  // 内存泄漏！
}
```

### 2. 重复释放

```c
int *arr = (int *)malloc(100 * sizeof(int));
free(arr);
free(arr);  // 错误！重复释放
```

### 3. 野指针

```c
int *p = (int *)malloc(sizeof(int));
free(p);
// p 现在是野指针
*p = 10;  // 未定义行为！
```

### 4. 栈上释放

```c
int arr[10];
free(arr);  // 错误！arr 是栈上分配的
```

### 5. 未检查 malloc 返回值

```c
int *arr = (int *)malloc(1000000000 * sizeof(int));
// 未检查 arr 是否为 NULL
arr[0] = 1;  // 如果分配失败，会导致段错误
```

### 6. 使用已释放的内存

```c
int *arr = (int *)malloc(10 * sizeof(int));
free(arr);
arr[0] = 10;  // 错误！使用已释放的内存
```

## 内存分配最佳实践

1. **始终检查 malloc/calloc/realloc 的返回值**
2. **malloc 后必须 free**
3. **free 后将指针设为 NULL**
4. **避免重复释放**
5. **避免内存泄漏**
6. **使用工具检测内存问题**（Valgrind、AddressSanitizer）
7. **配对使用 malloc/free**
