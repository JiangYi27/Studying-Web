# 线性结构

## 静态/动态顺序表

### 静态顺序表

静态顺序表使用固定大小的数组存储数据。

```c
#define MAX_SIZE 100

typedef struct {
    int data[MAX_SIZE];
    int size;
} SeqList;

void init_list(SeqList *list) {
    list->size = 0;
}

int insert(SeqList *list, int index, int value) {
    if (list->size >= MAX_SIZE || index < 0 || index > list->size) {
        return -1;  // 错误
    }

    // 移动元素
    for (int i = list->size; i > index; i--) {
        list->data[i] = list->data[i - 1];
    }

    list->data[index] = value;
    list->size++;
    return 0;  // 成功
}

int delete(SeqList *list, int index, int *value) {
    if (index < 0 || index >= list->size) {
        return -1;  // 错误
    }

    *value = list->data[index];

    // 移动元素
    for (int i = index; i < list->size - 1; i++) {
        list->data[i] = list->data[i + 1];
    }

    list->size--;
    return 0;  // 成功
}
```

### 动态顺序表

动态顺序表使用动态分配的内存存储数据，可以动态扩容。

```c
#include <stdlib.h>

typedef struct {
    int *data;
    int size;
    int capacity;
} DynamicSeqList;

DynamicSeqList *create_list(int initial_capacity) {
    DynamicSeqList *list = (DynamicSeqList *)malloc(sizeof(DynamicSeqList));
    if (list == NULL) {
        return NULL;
    }

    list->data = (int *)malloc(initial_capacity * sizeof(int));
    if (list->data == NULL) {
        free(list);
        return NULL;
    }

    list->size = 0;
    list->capacity = initial_capacity;
    return list;
}

int resize(DynamicSeqList *list, int new_capacity) {
    int *new_data = (int *)realloc(list->data, new_capacity * sizeof(int));
    if (new_data == NULL) {
        return -1;  // 扩容失败
    }

    list->data = new_data;
    list->capacity = new_capacity;
    return 0;
}

int push_back(DynamicSeqList *list, int value) {
    if (list->size >= list->capacity) {
        if (resize(list, list->capacity * 2) != 0) {
            return -1;  // 扩容失败
        }
    }

    list->data[list->size++] = value;
    return 0;
}

void destroy_list(DynamicSeqList *list) {
    if (list != NULL) {
        free(list->data);
        free(list);
    }
}
```

## 单链表

### 基本结构

```c
typedef struct Node {
    int data;
    struct Node *next;
} Node;
```

### 创建节点

```c
Node *create_node(int data) {
    Node *node = (Node *)malloc(sizeof(Node));
    if (node != NULL) {
        node->data = data;
        node->next = NULL;
    }
    return node;
}
```

### 在头部插入

```c
Node *insert_head(Node *head, int data) {
    Node *node = create_node(data);
    if (node != NULL) {
        node->next = head;
        head = node;
    }
    return head;
}
```

### 在尾部插入

```c
Node *insert_tail(Node *head, int data) {
    Node *node = create_node(data);
    if (node == NULL) {
        return head;
    }

    if (head == NULL) {
        return node;
    }

    Node *current = head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = node;

    return head;
}
```

### 删除节点

```c
Node *delete_node(Node *head, int data) {
    if (head == NULL) {
        return NULL;
    }

    // 删除头节点
    if (head->data == data) {
        Node *new_head = head->next;
        free(head);
        return new_head;
    }

    // 删除中间节点
    Node *current = head;
    while (current->next != NULL && current->next->data != data) {
        current = current->next;
    }

    if (current->next != NULL) {
        Node *to_delete = current->next;
        current->next = to_delete->next;
        free(to_delete);
    }

    return head;
}
```

### 遍历链表

```c
void print_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        printf("%d -> ", current->data);
        current = current->next;
    }
    printf("NULL\n");
}
```

### 释放链表

```c
void free_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
}
```

## 双链表

### 基本结构

```c
typedef struct DNode {
    int data;
    struct DNode *prev;
    struct DNode *next;
} DNode;
```

### 在头部插入

```c
DNode *dlist_insert_head(DNode *head, int data) {
    DNode *node = (DNode *)malloc(sizeof(DNode));
    if (node == NULL) {
        return head;
    }

    node->data = data;
    node->prev = NULL;
    node->next = head;

    if (head != NULL) {
        head->prev = node;
    }

    return node;
}
```

### 在尾部插入

```c
DNode *dlist_insert_tail(DNode *head, int data) {
    DNode *node = (DNode *)malloc(sizeof(DNode));
    if (node == NULL) {
        return head;
    }

    node->data = data;
    node->next = NULL;

    if (head == NULL) {
        node->prev = NULL;
        return node;
    }

    DNode *current = head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = node;
    node->prev = current;

    return head;
}
```

## 栈与队列

### 栈（后进先出，LIFO）

#### 基于数组的栈

```c
#define STACK_SIZE 100

typedef struct {
    int data[STACK_SIZE];
    int top;
} Stack;

void stack_init(Stack *s) {
    s->top = -1;
}

int stack_push(Stack *s, int value) {
    if (s->top >= STACK_SIZE - 1) {
        return -1;  // 栈满
    }
    s->data[++s->top] = value;
    return 0;
}

int stack_pop(Stack *s, int *value) {
    if (s->top < 0) {
        return -1;  // 栈空
    }
    *value = s->data[s->top--];
    return 0;
}

int stack_peek(Stack *s, int *value) {
    if (s->top < 0) {
        return -1;  // 栈空
    }
    *value = s->data[s->top];
    return 0;
}

int stack_is_empty(Stack *s) {
    return s->top == -1;
}
```

#### 基于链表的栈

```c
typedef struct StackNode {
    int data;
    struct StackNode *next;
} StackNode;

typedef struct {
    StackNode *top;
} LinkedStack;

void lstack_init(LinkedStack *s) {
    s->top = NULL;
}

int lstack_push(LinkedStack *s, int data) {
    StackNode *node = (StackNode *)malloc(sizeof(StackNode));
    if (node == NULL) {
        return -1;
    }
    node->data = data;
    node->next = s->top;
    s->top = node;
    return 0;
}

int lstack_pop(LinkedStack *s, int *value) {
    if (s->top == NULL) {
        return -1;
    }
    StackNode *top = s->top;
    *value = top->data;
    s->top = top->next;
    free(top);
    return 0;
}
```

### 队列（先进先出，FIFO）

#### 基于数组的队列（循环队列）

```c
#define QUEUE_SIZE 100

typedef struct {
    int data[QUEUE_SIZE];
    int front;
    int rear;
    int size;
} Queue;

void queue_init(Queue *q) {
    q->front = 0;
    q->rear = 0;
    q->size = 0;
}

int queue_enqueue(Queue *q, int value) {
    if (q->size >= QUEUE_SIZE) {
        return -1;  // 队列满
    }
    q->data[q->rear] = value;
    q->rear = (q->rear + 1) % QUEUE_SIZE;
    q->size++;
    return 0;
}

int queue_dequeue(Queue *q, int *value) {
    if (q->size <= 0) {
        return -1;  // 队列空
    }
    *value = q->data[q->front];
    q->front = (q->front + 1) % QUEUE_SIZE;
    q->size--;
    return 0;
}
```

#### 基于链表的队列

```c
typedef struct QNode {
    int data;
    struct QNode *next;
} QNode;

typedef struct {
    QNode *front;
    QNode *rear;
} LinkedQueue;

void lqueue_init(LinkedQueue *q) {
    q->front = NULL;
    q->rear = NULL;
}

int lqueue_enqueue(LinkedQueue *q, int data) {
    QNode *node = (QNode *)malloc(sizeof(QNode));
    if (node == NULL) {
        return -1;
    }
    node->data = data;
    node->next = NULL;

    if (q->rear == NULL) {
        q->front = q->rear = node;
    } else {
        q->rear->next = node;
        q->rear = node;
    }

    return 0;
}

int lqueue_dequeue(LinkedQueue *q, int *value) {
    if (q->front == NULL) {
        return -1;
    }
    QNode *front = q->front;
    *value = front->data;
    q->front = front->next;
    if (q->front == NULL) {
        q->rear = NULL;
    }
    free(front);
    return 0;
}
```

### 栈与队列的应用场景

栈和队列是两种最常用的受限线性表，应用十分广泛：

**栈（后进先出）的典型应用**

- **函数调用**：系统用栈保存函数调用的返回地址、局部变量，函数嵌套调用形成"调用栈"
- **括号匹配**：编译器中检查 `( ) [ ] { }` 是否配对，遇到左括号入栈，右括号出栈配对
- **表达式求值**：中缀表达式转后缀表达式（逆波兰式）并计算
- **递归的实现**：递归调用本质上就是栈的压入与弹出过程

**队列（先进先出）的典型应用**

- **任务调度**：操作系统按到达顺序调度进程/线程
- **消息队列**：生产者-消费者模型中缓冲消息
- **广度优先搜索（BFS）**：图的层序扩展依赖队列
- **打印机任务排队**：多个打印任务按提交顺序依次执行

**如何选择线性结构**

- 需要随机访问某个位置：选**顺序表**
- 需要频繁在中间插入、删除：选**链表**
- 只在两端操作、后进先出：选**栈**
- 先进先出：选**队列**

## 线性结构对比

| 结构 | 访问 | 插入 | 删除 | 空间 |
|------|------|------|------|------|
| 顺序表 | O(1) | O(n) | O(n) | 连续 |
| 单链表 | O(n) | O(1) | O(1) | 分散 |
| 双链表 | O(n) | O(1) | O(1) | 分散 |
| 栈 | O(1) | O(1) | O(1) | 连续/分散 |
| 队列 | O(1) | O(1) | O(1) | 连续/分散 |
