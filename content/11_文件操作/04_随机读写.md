# 随机读写

## 文件定位

### fseek() — 移动文件位置指针

```c
int fseek(FILE *stream, long offset, int whence);
```

### 起始位置

| 常量 | 值 | 说明 |
|------|-----|------|
| `SEEK_SET` | 0 | 文件开头 |
| `SEEK_CUR` | 1 | 当前位置 |
| `SEEK_END` | 2 | 文件末尾 |

### 示例

```c
FILE *fp = fopen("data.txt", "r");
if (fp != NULL) {
    // 移动到文件开头
    fseek(fp, 0, SEEK_SET);

    // 移动到当前位置后 10 个字节
    fseek(fp, 10, SEEK_CUR);

    // 移动到文件末尾前 5 个字节
    fseek(fp, -5, SEEK_END);

    fclose(fp);
}
```

### fseek 的用途

1. **随机访问文件**：直接跳转到文件的任意位置
2. **重新读取文件**：将位置指针移回开头
3. **追加数据前定位**：移动到文件末尾

## ftell() — 获取当前位置

### 基本语法

```c
long ftell(FILE *stream);
```

返回文件位置指针相对于文件开头的偏移量（以字节为单位）。

### 示例

```c
FILE *fp = fopen("data.txt", "r");
if (fp != NULL) {
    // 获取当前位置
    long pos1 = ftell(fp);  // 0（文件开头）

    // 移动位置
    fseek(fp, 50, SEEK_SET);

    // 获取新位置
    long pos2 = ftell(fp);  // 50

    // 计算文件大小
    fseek(fp, 0, SEEK_END);
    long file_size = ftell(fp);

    printf("文件大小: %ld 字节\n", file_size);

    fclose(fp);
}
```

## rewind() — 重置到文件开头

### 基本语法

```c
void rewind(FILE *stream);
```

将文件位置指针重置到文件开头，等价于 `fseek(stream, 0, SEEK_SET)`。

### 示例

```c
FILE *fp = fopen("data.txt", "r");
if (fp != NULL) {
    // 读取第一遍
    char buf[100];
    while (fgets(buf, sizeof(buf), fp) != NULL) {
        printf("第一遍: %s", buf);
    }

    // 重置到开头
    rewind(fp);

    // 读取第二遍
    while (fgets(buf, sizeof(buf), fp) != NULL) {
        printf("第二遍: %s", buf);
    }

    fclose(fp);
}
```

## 随机读写的应用

### 随机访问二进制文件

```c
#include <stdio.h>

struct Record {
    int id;
    char name[50];
    float score;
};

int main(void) {
    // 写入记录
    struct Record records[] = {
        {1, "Alice", 95.5},
        {2, "Bob", 88.0},
        {3, "Charlie", 92.5}
    };

    FILE *fp = fopen("records.dat", "wb");
    if (fp != NULL) {
        fwrite(records, sizeof(struct Record), 3, fp);
        fclose(fp);
    }

    // 随机读取第二条记录
    fp = fopen("records.dat", "rb");
    if (fp != NULL) {
        struct Record rec;

        // 移动到第二条记录的位置
        fseek(fp, 1 * sizeof(struct Record), SEEK_SET);

        // 读取记录
        fread(&rec, sizeof(struct Record), 1, fp);
        printf("ID: %d, Name: %s, Score: %.1f\n", rec.id, rec.name, rec.score);

        fclose(fp);
    }

    return 0;
}
```

### 更新文件中的特定记录

```c
void update_record(const char *filename, int record_id, float new_score) {
    FILE *fp = fopen(filename, "r+b");  // 读写二进制模式
    if (fp == NULL) {
        perror("fopen");
        return;
    }

    struct Record rec;
    while (fread(&rec, sizeof(struct Record), 1, fp) == 1) {
        if (rec.id == record_id) {
            // 计算记录在文件中的位置
            long pos = ftell(fp) - sizeof(struct Record);

            // 更新分数
            rec.score = new_score;

            // 移动回记录位置并写入
            fseek(fp, pos, SEEK_SET);
            fwrite(&rec, sizeof(struct Record), 1, fp);
            break;
        }
    }

    fclose(fp);
}
```

### 文件复制（使用随机读写）

```c
#include <stdio.h>

void copy_file(const char *src, const char *dst) {
    FILE *fps = fopen(src, "rb");
    FILE *fpd = fopen(dst, "wb");

    if (fps == NULL || fpd == NULL) {
        perror("fopen");
        if (fps) fclose(fps);
        if (fpd) fclose(fpd);
        return;
    }

    // 获取文件大小
    fseek(fps, 0, SEEK_END);
    long size = ftell(fps);
    rewind(fps);

    // 分配缓冲区
    char *buf = (char *)malloc(size);
    if (buf == NULL) {
        fclose(fps);
        fclose(fpd);
        return;
    }

    // 读取并写入
    fread(buf, 1, size, fps);
    fwrite(buf, 1, size, fpd);

    free(buf);
    fclose(fps);
    fclose(fpd);
}
```

## 文件状态检查

### feof() — 检查文件结束

```c
int feof(FILE *stream);
```

当文件结束标志被设置时返回非零值。

```c
FILE *fp = fopen("data.txt", "r");
if (fp != NULL) {
    int ch;
    while (!feof(fp)) {
        ch = fgetc(fp);
        if (ch != EOF) {
            putchar(ch);
        }
    }
    fclose(fp);
}
```

### ferror() — 检查错误

```c
int ferror(FILE *stream);
```

当错误标志被设置时返回非零值。

### clearerr() — 清除错误标志

```c
void clearerr(FILE *stream);
```

清除文件结束标志和错误标志。

### 正确读取文件的方式

```c
// 推荐方式：检查 fread/fgets 的返回值
char buf[100];
while (fgets(buf, sizeof(buf), fp) != NULL) {
    // 处理 buf
}

// 不推荐方式：使用 feof
while (!feof(fp)) {
    fgets(buf, sizeof(buf), fp);  // 可能在最后一次读取后多处理一次
}
```

## 随机读写的注意事项

1. **二进制模式使用 `r+b` 或 `w+b`**
2. **文本模式的随机读写可能不可靠**（换行符转换）
3. **使用 `fseek` 前检查文件是否打开成功**
4. **使用 `ftell` 计算文件大小时，先 `fseek` 到末尾**
5. **`rewind` 等价于 `fseek(fp, 0, SEEK_SET)`**
6. **更新文件记录时使用 `r+b` 模式**
